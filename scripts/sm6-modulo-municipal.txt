  //─ 4. dados. Cada carregador devolve {linhas, total, escolasDistintas, erro}: `escolasDistintas` é a
  //    medida do ESCOPO EFETIVO que o backend deixou ver, e é impressa sempre que difere do pedido.
  const cache = {};
  // A escola duma linha pode estar na própria coluna, num embutido, ou um nível acima: as FKs medidas em
  // evidence/sm6-colunas-e-vinculos-17-11.json mostram que `bfr_lesson_attendance` NÃO tem school_id —
  // a escola só chega por `students.school_id` (e `supervision_requests.school_id` é chave de
  // `sports_schools`, não de `schools`). Quem não sabe isto escreve um select que o backend recusa.
  const escolaDe = (r) => (!r ? '' : r.school_id || (r.schools && r.schools.id) || (r.sports_schools && r.sports_schools.id)
    || (r.students && r.students.school_id) || (r.classes && r.classes.school_id)
    || (r.students && r.students.classes && r.students.classes.school_id) || '');
  async function ler(chave, build) {
    const c = client();
    if (!c) return { linhas: [], total: 0, escolasDistintas: 0, erro: 'sem cliente Supabase (BFR_APP_RUNTIME.client)' };
    if (cache[chave]) return cache[chave];
    estado.carregando[chave] = true; desenharCabeca();
    try {
      const q = build(c.from.bind(c));
      const { data: rows, error, count } = await q;
      const linhas = Array.isArray(rows) ? rows : (rows ? [rows] : []);
      const out = {
        linhas, erro: error ? (error.message || error.code || 'erro') : '', total: count == null ? linhas.length : count,
        escolasDistintas: new Set(linhas.map(escolaDe).filter(Boolean)).size
      };
      if (out.erro) estado.erros[chave] = out.erro;
      cache[chave] = out; return out;
    } catch (e) {
      const out = { linhas: [], total: 0, escolasDistintas: 0, erro: String((e && e.message) || e) };
      estado.erros[chave] = out.erro; return out;
    } finally { estado.carregando[chave] = false; }
  }

O DDL candidato <code>scripts/sm4-ddl-municipal-candidata.sql</code> é o que fecha a diferença.</div>'
      : ''));

  function painel(d) {
    const es = d.escolas.linhas || [];

    const al = d.alunos.linhas, tur = d.turmas.linhas, prof = d.profissionais.linhas;
    const porCidade = {}; for (const s of es) porCidade[s.city || 'sem cidade'] = (porCidade[s.city || 'sem cidade'] || 0) + 1;
    const porEscola = {}; for (const a of al) { const n = (a.schools && a.schools.name) || a.school_id || 'sem escola'; porEscola[n] = (porEscola[n] || 0) + 1; }
    // `closed_at` é o que se mede; não adivinho a grafia do enum `status`. Os valores existentes de
    // status saem como etiquetas, para o utilizador ver o que a base realmente guarda.

    // O mesmo critério: a taxa de presença só é calculada se existir um valor de `status` que se
    // reconheça como presença; senão o cartão mostra o que há, não uma percentagem inventada.
    const rotulos = {}; for (const f of d.freq.linhas) { const s = String(f.status || 'sem situação'); rotulos[s] = (rotulos[s] || 0) + 1; }
    const rotulosPres = Object.keys(rotulos).filter((s) => /pres/i.test(s));
    const pres = d.freq.linhas.filter((f) => rotulosPres.includes(String(f.status || ''))).length;
    const taxa = (d.freq.linhas.length && rotulosPres.length) ? Math.round((pres / d.freq.linhas.length) * 100) + '%' : '—';

      + Object.entries(rotulos).slice(0, 8).map(([k, v]) => '<span class="mun-chip">chamada: ' + esc(k) + ' · ' + v + '</span>').join(' ')
      + (!Object.keys(situacoes).length && !Object.keys(rotulos).length ? '<div class="mun-empty">Nenhuma linha de processo ou chamada devolvida por este acesso.</div>' : '')
      + '</div></div>'

  function pedagogico(d) {
    const termos = {}; const porDisciplina = {}; const porEscola = {};
    for (const g of d.linhas) {
      const t = g.term == null ? 'sem período' : String(g.term);
      (termos[t] = termos[t] || []).push(Number(g.score));
      const s = g.subject || 'sem disciplina';
      (porDisciplina[s] = porDisciplina[s] || []).push(Number(g.score));
      const es = (g.students && g.students.school_id) || 'sem escola';
      (porEscola[es] = porEscola[es] || []).push(Number(g.score));
    }
    const med = (a) => (a.length ? a.reduce((x, y) => x + y, 0) / a.length : null);
    const fmt = (v) => (v == null ? '—' : v.toFixed(2).replace('.', ','));
    const linhasT = Object.keys(termos).sort().map((t) => '<tr><td>' + esc(t) + '</td><td>' + num(termos[t].length) + '</td><td><b>' + fmt(med(termos[t])) + '</b></td><td>' + fmt(Math.min(...termos[t])) + '</td><td>' + fmt(Math.max(...termos[t])) + '</td></tr>').join('');
    const linhasD = Object.keys(porDisciplina).sort().map((s) => '<tr><td>' + esc(s) + '</td><td>' + num(porDisciplina[s].length) + '</td><td><b>' + fmt(med(porDisciplina[s])) + '</b></td></tr>').join('');
    const nomeEscola = (id) => { const e = (cache.escolas_todo || { linhas: [] }).linhas.find((x) => x.id === id); return e ? e.name : (id === 'sem escola' ? id : String(id).slice(0, 8)); };
    const linhasE = Object.keys(porEscola).sort((a, b) => med(porEscola[b]) - med(porEscola[a])).map((id) => '<tr><td>' + esc(nomeEscola(id)) + '</td><td>' + num(porEscola[id].length) + '</td><td><b>' + fmt(med(porEscola[id])) + '</b></td><td>' + fmt(Math.min(...porEscola[id])) + '</td></tr>').join('');
    return '<div class="mun-panel"><h3>Acompanhamento pedagógico — por período (campo `term` de grades)</h3>'
      + (d.linhas.length ? '<div class="mun-scroll"><table><thead><tr><th>Período</th><th>Avaliações</th><th>Média</th><th>Mín.</th><th>Máx.</th></tr></thead><tbody>' + linhasT + '</tbody></table></div>'
        : '<div class="mun-empty">Nenhuma nota devolvida por este acesso.</div>')
      + (linhasD ? '<h3 style="margin-top:16px">Por disciplina</h3><div class="mun-scroll"><table><thead><tr><th>Disciplina</th><th>Registros</th><th>Média</th></tr></thead><tbody>' + linhasD + '</tbody></table></div>' : '')
      + (linhasE ? '<h3 style="margin-top:16px">Por unidade escolar (via aluno)</h3><div class="mun-scroll"><table><thead><tr><th>Escola</th><th>Registros</th><th>Média</th><th>Mín.</th></tr></thead><tbody>' + linhasE + '</tbody></table></div><p class="s" style="font-size:11.5px;color:#6b7a89">A escola não está em `grades`: chega pelo caminho grades → students → school_id, que é a única FK medida para estas tabelas.</p>' : '')
      + '<p class="s" style="font-size:11.5px;color:#6b7a89">Os quatro bimestres do §4.5 só aparecem se existirem lançamentos: a tabela `grades` devolve ' + num(d.linhas.length) + ' linhas com ' + Object.keys(termos).length + ' valor(es) de `term`. Nenhum período é inventado para preencher a escala.</p>'

  function frequencia(d) {
    // Os valores de `status` vêm do banco, não da minha cabeça: primeiro faço o censo dos rótulos, depois
    // classifico como falta apenas os rótulos que literally contêm «falt». Se a base usar outra grafia, o
    // painel mostra os rótulos medidos em vez de fingir uma percentagem.
    const rotulos = {}; for (const f of d.linhas) { const s = String(f.status || 'sem situação'); rotulos[s] = (rotulos[s] || 0) + 1; }
    const rotFalta = Object.keys(rotulos).filter((s) => /falt/i.test(s));
    const porAluno = {};
    for (const f of d.linhas) {
      const k = f.student_id || 'sem aluno';
      const o = (porAluno[k] = porAluno[k] || { nome: (f.students && f.students.official_name) || '—', total: 0, falta: 0, ultima: '' });
      o.total++;
      if (rotFalta.includes(String(f.status || ''))) o.falta++;
      if (f.lesson_date && (!o.ultima || f.lesson_date > o.ultima)) o.ultima = f.lesson_date;
    }
    const list = Object.values(porAluno).filter((o) => o.total).sort((a, b) => (b.falta / b.total) - (a.falta / a.total));
    const risco = list.filter((o) => o.total >= 4 && o.falta / o.total >= 0.25);
    const alerts = d.alertas || { linhas: [], erro: 'não lido' };
    return '<div class="mun-panel"><h3>Frequência e risco de abandono</h3>'
      + '<div class="mun-cards">' + cartao('Chamadas registradas', num(d.linhas.length), 'bfr_lesson_attendance')
      + cartao('Alunos com chamada', num(list.length), Object.keys(rotulos).length + ' valor(es) de status: ' + esc(Object.keys(rotulos).slice(0, 6).join(', ') || '—'))
      + cartao('Acima de 25% de ausências', rotFalta.length ? num(risco.length) : '—', rotFalta.length ? 'rótulos tratados como ausência: ' + rotFalta.join(' / ') : 'nenhum rótulo de ausência na base — percentagem não calculável') + '</div>'
      + '<div class="mun-warn"><strong>Nenhum aluno é classificado como evadido aqui.</strong> §4.6 do briefing: falta não é evasão. A situação de acompanhamento vive em `supervision_alerts` — ' + num(alerts.linhas.length) + ' registro(s) devolvido(s) por este acesso' + (alerts.erro ? ' (leitura com erro: ' + esc(alerts.erro) + ')' : '') + ', e só se altera por decisão humana.</div>'
      + (list.length ? '<div class="mun-scroll"><table><thead><tr><th>Aluno</th><th>Chamadas</th><th>Faltas</th><th>%</th><th>Última data</th></tr></thead><tbody>'
        + list.slice(0, 200).map((o) => '<tr><td>' + esc(o.nome) + '</td><td>' + num(o.total) + '</td><td>' + num(o.falta) + '</td><td>' + (o.total ? Math.round((o.falta / o.total) * 100) + '%' : '—') + '</td><td>' + data(o.ultima) + '</td></tr>').join('')
        + '</tbody></table></div>' : '<div class="mun-empty">Sem chamadas devolvidas por este acesso.</div>')
      + avisoEscopo(d, (cache.escolas_todo || { linhas: [] }).linhas.length) + '</div>';
  }

  function corpo(d) {
    if (estado.detalhe && RENDER[estado.tab] === escolas) return detalheEscola(estado.detalhe);
    const fn = RENDER[estado.tab];
    if (!fn) return '<div class="mun-panel"><div class="mun-empty">Selecione uma seção.</div></div>';

    if (!d) return '<div class="mun-panel"><div class="mun-empty">Sem dados carregados.</div></div>';

  }

  function desenharCabeca() {
    const raiz = document.getElementById(VIEW_ID); if (!raiz) return;
    const alvo = raiz.querySelector('[data-mun-conteudo]'); if (alvo) alvo.innerHTML = '<div class="mun-panel"><div class="mun-empty">Carregando dados reais…</div></div>';
  }
  async function desenhar(raiz) {
    estilo();
    const permissao = aptoParaSessao();
    if (!permissao.apto) {
      raiz.innerHTML = cabecalho() + '<div class="mun-panel"><div class="mun-warn"><strong>Acesso não concedido a esta área.</strong> ' + esc(permissao.por) + '.</div>'
        + '<p style="font-size:13px">Chegou aqui por um caminho direto, e não por um botão que o menu lhe mostra. Nada foi carregado: a lista de escolas, alunos e processos continua a depender do que o backend autoriza — hoje pelo mapa de papéis, e depois por <code>bfr_is_municipal_authority()</code> quando o DDL SM4 for aplicado.</p></div>';
      raiz.dataset.munApto = 'nao';
      return;
    }
    raiz.dataset.munApto = 'sim';
    if (!raiz.dataset.munBuilt) {
      raiz.dataset.munBuilt = '1';
      raiz.innerHTML = cabecalho() + '<div class="mun-shell">' + lado() + '<div data-mun-conteudo></div></div>';
      ligarEventos(raiz);
    } else {
      raiz.querySelectorAll('[data-mun-tab]').forEach((b) => b.setAttribute('aria-current', b.dataset.munTab === estado.tab ? 'page' : 'false'));
    }
    const alvo = raiz.querySelector('[data-mun-conteudo]');
    const carregador = CARREGADORES[estado.tab];
    if (!carregador) { alvo.innerHTML = '<div class="mun-panel"><div class="mun-empty">Selecione uma seção.</div></div>'; return; }
    alvo.innerHTML = '<div class="mun-panel"><div class="mun-empty">Carregando dados reais…</div></div>';
    try {
      const d = await carregador();
      if (d && d.erro) estado.erros[estado.tab] = d.erro;
      alvo.innerHTML = '<div data-mun-sonda></div>' + corpo(d);
      pintarSonda();
      if (estado.erros[estado.tab]) alvo.querySelector('[data-mun-sonda]').insertAdjacentHTML('afterend', '<div class="mun-warn"><strong>Erro anterior nesta seção:</strong> ' + esc(estado.erros[estado.tab]) + '</div>');
    } catch (e) {
      alvo.innerHTML = '<div class="mun-panel"><div class="mun-warn"><strong>Falha ao carregar:</strong> ' + esc(String((e && e.message) || e)) + '</div></div>';
    }
  }

  function csv(nome, rows) {
    const cols = [...new Set(rows.flatMap((r) => Object.keys(r).filter((k) => typeof r[k] !== 'object' || r[k] === null)))];
    const val = (v) => (v && typeof v === 'object' ? JSON.stringify(v) : String(v == null ? '' : v));
    const corpo = [cols.join(';'), ...rows.map((r) => cols.map((c) => '"' + val(r[c]).replace(/"/g, '""') + '"').join(';'))].join('\n');
    const url = URL.createObjectURL(new Blob(['\ufeff' + corpo], { type: 'text/csv;charset=utf-8' }));
    const a = document.createElement('a'); a.href = url; a.download = 'bfr-municipal-' + nome + '-' + new Date().toISOString().slice(0, 10) + '.csv';
    document.body.appendChild(a); a.click(); a.remove(); setTimeout(() => URL.revokeObjectURL(url), 4000);
  }

  function ligarEventos(raiz) {
    raiz.addEventListener('click', (ev) => {
      const tab = ev.target.closest('[data-mun-tab]');
      if (tab) { estado.tab = tab.dataset.munTab; estado.detalhe = ''; desenhar(raiz); return; }
      const escola = ev.target.closest('[data-mun-escola]');
      if (escola) { estado.detalhe = escola.dataset.munEscola; desenhar(raiz); return; }
      if (ev.target.closest('[data-mun-volta]')) { estado.detalhe = ''; desenhar(raiz); return; }
      const rec = ev.target.closest('[data-mun-recarregar]');
      if (rec) { delete cache['alunos_' + [estado.escola, estado.busca, estado.ano, estado.turno].join('|')]; desenhar(raiz); return; }
      if (ev.target.closest('[data-mun-print]')) { window.print(); return; }
      const exp = ev.target.closest('[data-mun-export]');
      if (exp) {
        const que = exp.dataset.munExport;
        if (que === 'censo') csv('censo', (cache.escolas_todo || { linhas: [] }).linhas.map((e) => ({ ...e, alunos_visiveis: contaAlunos(e.id) })));
        if (que === 'alunos') csv('alunos', (cache.alunos_todo || { linhas: [] }).linhas);
        return;
      }
      const cacheLimpar = ev.target.closest('[data-mun-cache]');
      if (cacheLimpar) { Object.keys(cache).forEach((k) => delete cache[k]); desenhar(raiz); }
    });
    raiz.addEventListener('change', (ev) => {
      const f = ev.target.closest('[data-mun-filtro]'); if (!f) return;
      const tipo = f.dataset.munFiltro;
      if (tipo === 'escola') estado.escola = f.value;
      if (tipo === 'busca') estado.busca = f.value;
      if (tipo === 'tipo') estado.fEscolaTipo = f.value;
      desenhar(raiz);
    });
  }

  //─ 6. entrada. Espera o runtime do bundle; sem runtime, sem cliente, sem suposições.
  let iniciado = false;
  function aptoParaSessao() {
    const rt = RT(); const papel = (rt.profile || {}).role || '';
    if (!papel) return { apto: false, por: 'sem perfil carregado' };
    const view = viewRoleDe(papel);
    const map = rt.roleSectionAccess || {};
    // A mesma regra do bundle, e não uma versão minha: `admin` recebe o mapa de admin + psicologia +
    // multiprofissional; os restantes lêem `roleSectionAccess[viewRole]`, com `parent` por omissão.
    const permitido = view === 'admin' ? [...new Set([...(map.admin || []), 'psychology', 'multiprofissional'])]
      : (map[view] || map.parent || []);
    return { apto: permitido.includes(SECT), por: 'papel=' + papel + ' viewRole=' + view + ' no mapa=' + permitido.includes(SECT) };
  }
  // §5/§6 medidos a 18/11: um papel que não tem chave em `roleSectionAccess` (o `secretary` real é o
  // caso) não é escondido pelo shell, porque o shell não tem lista nenhuma para ele — e a entrada
  // municipal ficava visível, levando a um clique que abre apenas a faixa de negação. Esconder aqui é
  // apresentação; a decisão continua em `aptoParaSessao()` e na sonda do backend, e nenhum dos dois
  // depende desta visibilidade. O intervalo não tem tecto: o seletor de papel do shell volta a desenhar
  // a sidebar a cada troca, e um caputo deixaria a entrada reaparecer depois dele.
  function aplicarVisibilidade() {
    const veredito = aptoParaSessao();
    for (const b of document.querySelectorAll('[data-bfr-municipal-nav]')) {
      b.classList.toggle('hidden', !veredito.apto);
      b.style.display = veredito.apto ? '' : 'none';
      b.dataset.munApto = veredito.apto ? 'sim' : 'nao';
    }
    const raiz = document.getElementById(VIEW_ID);
    if (raiz) raiz.dataset.munApto = veredito.apto ? 'sim' : 'nao';
    return veredito;
  }
  const tentativas = { registos: 0, limite: 50 };
  function garantir() {
    if (!document.querySelector('#appShell main')) return false;
    registar(); estilo();
    const vistaOk = !!document.getElementById(VIEW_ID);
    const navOk = !!document.querySelector('.sidebar [data-bfr-municipal-nav]');
    if (vistaOk && navOk) aplicarVisibilidade();
    return vistaOk && navOk;
  }
  function tentar() {
    if (iniciado) return;
    if (!RT().client) return;
    iniciado = true;
    document.addEventListener('click', (ev) => {
      const nav = ev.target.closest('[data-section="' + SECT + '"]');
      if (!nav) return;
      ev.preventDefault(); ev.stopImmediatePropagation();
      abrir();
    }, true);
    // O shell desenha a sidebar dele próprio, e `hideLegacy()` passa por cima: por isso o registo é
    // re-tentado até a vista E o botão existirem, e volta a tentar-se a cada navegação.
    const ciclo = setInterval(() => {
      tentativas.registos++;
      const pronto = garantir();
      if (pronto || tentativas.registos >= tentativas.limite) clearInterval(ciclo);
      if (pronto) aplicarVisibilidade();
    }, 400);
    setTimeout(() => clearInterval(ciclo), 30000);
    const vigilia = setInterval(() => { if (garantir()) aplicarVisibilidade(); }, 600);
    sondarBackend();
  }
  const timer = setInterval(() => tentar(), 250);
  setTimeout(() => clearInterval(timer), 30000);
  window.BFR_MUNICIPAL = { abrir, registar, garantir, aptoParaSessao, aplicarVisibilidade, sondarBackend, sonda, estado, cache, SECT, PAPEIS_MUNICIPAIS };
})();
